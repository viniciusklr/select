(async () => {
  const PRIORITY = "1";
  const WAIT_MOVE_MS = 3500;
  const WAIT_PRIORITY_MS = 2500;
  const WAIT_SAVE_MS = 4000;

  const raw = prompt("Nomes EXATOS (um por linha):", "") || "";
  const names = raw.split(/\r?\n/).map((n) => n.trim()).filter(Boolean);
  if (!names.length) return console.warn("Nenhum nome informado.");

  const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

  const norm = (s) =>
    (s || "")
      .normalize("NFD")
      .replace(/[\u0300-\u036f]/g, "")
      .toLowerCase()
      .replace(/\s+/g, " ")
      .trim();

  const setValue = (el, value) => {
    const setter = Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, "value")?.set;
    if (setter) setter.call(el, value);
    else el.value = value;
    el.dispatchEvent(new Event("input", { bubbles: true }));
    el.dispatchEvent(new Event("change", { bubbles: true }));
    el.dispatchEvent(new Event("blur", { bubbles: true }));
  };

  const pageClickById = (id) => {
    const s = document.createElement("script");
    s.textContent = `
      (function () {
        var el = document.getElementById(${JSON.stringify(id)});
        if (!el) throw new Error("Elemento não encontrado: ${id}");
        el.disabled = false;
        el.click();
      })();
    `;
    document.documentElement.appendChild(s);
    s.remove();
  };

  const getRows = () =>
    [...document.querySelectorAll('input[id="fakeButtonAgentAssigned__Name"]')].map((nameInput) => {
      let cur = nameInput.parentElement;
      for (let i = 0; i < 10 && cur; i++) {
        if (cur.querySelector('input[id="TextBoxAgentAssigned_EditPriority"]')) break;
        cur = cur.parentElement;
      }
      return {
        nameValue: (nameInput.value || "").trim(),
        rowText: (cur?.innerText || "").replace(/\s+/g, " ").trim(),
        check: cur?.querySelector('input[id="CheckBoxAssignedAgent_Selected"]'),
        priorityInput: cur?.querySelector('input[id="TextBoxAgentAssigned_EditPriority"]'),
      };
    });

  const matchRow = (row, wantedNorm) => {
    const a = norm(row.nameValue);
    const b = norm(row.rowText);
    return a === wantedNorm || b === wantedNorm || b.startsWith(wantedNorm + " ");
  };

  const rdo = document.getElementById(
    "ctl00_PageContent_TabContainer1_TabPanelAgents_rdoButtonUnrelated"
  );
  if (rdo && !rdo.checked) rdo.click();

  let list = document.getElementById("ListBoxUnrelatedAgents");
  const sendBtnId = "ctl00_PageContent_TabContainer1_TabPanelAgents_SendRelatedAgents";
  const saveBtnId = "ctl00_PageContent_ButtonEditCampaign";

  if (!list || !document.getElementById(sendBtnId) || !document.getElementById(saveBtnId)) {
    return console.error("Abra a aba Agentes da campanha.");
  }

  for (const o of list.options) o.selected = false;

  const toMove = [];
  const already = [];
  const notFound = [];
  const before = getRows();

  for (const name of names) {
    const n = norm(name);

    if (before.some((r) => matchRow(r, n))) {
      already.push(name);
      console.log("⏭️ já relacionado:", name);
      continue;
    }

    const opt = [...list.options].find((o) => norm(o.text) === n);
    if (!opt) {
      notFound.push(name);
      console.warn("❌ não encontrado:", name);
      continue;
    }

    opt.selected = true;
    toMove.push(opt.text);
    console.log("✔ selecionado:", opt.text);
  }

  if (toMove.length) {
    list.focus();
    list.dispatchEvent(new Event("change", { bubbles: true }));
    console.log("➡️ Movendo", toMove.length, "agente(s)...");
    pageClickById(sendBtnId);
    await sleep(WAIT_MOVE_MS);

    const afterNames = new Set(getRows().map((r) => norm(r.nameValue || r.rowText)));
    const confirmed = toMove.filter((t) => afterNames.has(norm(t)) || [...afterNames].some((a) => a.startsWith(norm(t) + " ")));
    console.log("✅ Confirmados nos relacionados:", confirmed);
  }

  if (notFound.length) console.warn("Não encontrados:", notFound);

  const targets = [...new Set([...toMove, ...already].map(norm))];
  if (!targets.length) return console.error("Nenhum agente alvo.");

  for (const row of getRows()) {
    if (row.check?.checked) row.check.click();
  }
  await sleep(200);

  let marked = 0;
  for (const row of getRows()) {
    if (!targets.some((t) => matchRow(row, t))) continue;
    if (row.check && !row.check.checked) row.check.click();
    marked++;
    console.log("☑️", row.nameValue || row.rowText);
  }

  if (!marked) {
    console.error("Não achei checkboxes dos agentes. Relacionados:", getRows().map((r) => r.nameValue || r.rowText));
    return;
  }

  const bulkInput = document.getElementById("TextBoxAssignAgentMenu_Priorities_Set");
  const bulkSetId = "ctl00_PageContent_TabContainer1_TabPanelAgents_LabelAssignAgentMenu_Priorities_Set";

  if (!bulkInput || !document.getElementById(bulkSetId)) {
    return console.error("Controles de prioridade em lote não encontrados.");
  }

  bulkInput.focus();
  setValue(bulkInput, PRIORITY);
  if (bulkInput.value !== PRIORITY) bulkInput.value = PRIORITY;
  console.log("🎯 Prioridade lote =", bulkInput.value);

  pageClickById(bulkSetId);
  await sleep(WAIT_PRIORITY_MS);

  for (const row of getRows()) {
    if (!targets.some((t) => matchRow(row, t))) continue;
    console.log("Prioridade na tela:", row.nameValue || row.rowText, "=>", row.priorityInput?.value);
  }

  // ========== 3) SALVAR ==========
  console.log("💾 Salvando...");
  pageClickById(saveBtnId);
  await sleep(WAIT_SAVE_MS);

  console.log("✅ Finalizado.", {
    pedidos: names.length,
    movidos: toMove.length,
    jaRelacionados: already.length,
    naoEncontrados: notFound.length,
    marcadosPrioridade: marked,
  });
})();
